import '@ahaslides-product/design/aha-card-select'; // registers <aha-card-select>
import { useRef, useEffect } from 'react';

// React 18 needs a thin wrapper; React 19 can use <aha-card-select> directly.
// options is a PROPERTY (array), so set it imperatively; the grid is one roving radiogroup.
function AhaCardSelect({ options, value, columns, iconOnly, tile, hideLabels, disabled, onChange, children, ...rest }) {
  const ref = useRef();
  useEffect(() => {
    const el = ref.current;
    el.options = options || [];
    el.value = value ?? '';   // '' checks no card (e.g. a custom theme); the first card keeps the tab stop
    el.tile = tile || 'default';   // 'preview' shows each option's slotted child as its preview
    el.hideLabels = !!hideLabels;   // preview tiles that already draw the name: label stays as aria-label only
    el.iconOnly = !!iconOnly;   // compact 44px squares; label kept as aria-label and shown as a tooltip on hover and keyboard focus
    el.disabled = !!disabled;
    const h = (e) => onChange?.(e.detail.value);   // composed CustomEvent
    el.addEventListener('change', h);
    return () => el.removeEventListener('change', h);
  }, [options, value, iconOnly, tile, hideLabels, disabled, onChange]);
  return <aha-card-select ref={ref} {...rest} {...(columns ? { columns: String(columns) } : {})}>{children}</aha-card-select>;
}

// usage: labelled tiles (default) or compact iconOnly squares
