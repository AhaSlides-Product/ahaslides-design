<!-- Paste-and-run: save as .html and open in a browser. No build step.
     <aha-card-select> is the SAME shared custom element React and Vue consume — here in its native
     form. Theming comes only from the --aha-* tokens in tokens.css. The whole grid is one radiogroup:
     Tab in, then Arrow keys move-and-select. Icons are summoned by name — never an inline <svg>. -->
<link rel="stylesheet" href="https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/tokens.css">
<script type="module">
  import 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/aha-card-select.js';   // registers <aha-card-select>
</script>

<!-- labelled tiles — stretch to fill each of `columns` tracks -->
<aha-card-select id="chart" columns="3" value="bar" style="max-width:360px"></aha-card-select>

<!-- iconOnly — compact 44px squares packed to the start; the label is kept as aria-label and shown as a tooltip on hover and keyboard focus -->
<aha-card-select id="chartCompact" icon-only columns="3" value="bar"></aha-card-select>

<!-- tile="preview": each option names a slot; the app draws the preview as a light-DOM child with that slot.
     hide-labels keeps the label as aria-label only, for previews that already show the name. An empty value checks no card. -->
<aha-card-select id="looks" tile="preview" hide-labels columns="3" value="classic" aria-label="Certificate design"
  style="max-width:480px; --aha-card-select-preview-ratio:297 / 210">
  <div slot="classic" style="height:100%; background:var(--aha-bg-layout)"></div>
  <div slot="midnight" style="height:100%; background:var(--aha-bg-dark)"></div>
  <div slot="blush" style="height:100%; background:var(--aha-color-primary)"></div>
</aha-card-select>

<script type="module">
  // options: value + label + an icon NAME from the DS library
  const options = [
    { value: 'bar',   label: 'Bar',   icon: 'system-chart-bar' },
    { value: 'line',  label: 'Line',  icon: 'system-chart-line-up' },
    { value: 'pie',   label: 'Pie',   icon: 'system-chart-pie' },
  ];
  const el = document.getElementById('chart');
  el.options = options;
  el.addEventListener('change', (e) => console.log('chart type:', e.detail.value));

  document.getElementById('chartCompact').options = options;

  document.getElementById('looks').options = [
    { value: 'classic',  label: 'Classic',  slot: 'classic' },
    { value: 'midnight', label: 'Midnight', slot: 'midnight' },
    { value: 'blush',    label: 'Blush',    slot: 'blush' },
  ];
</script>
