<!-- placement="bottomLeft" (default) | "bottomRight" | "topLeft" | "topRight"
     trigger="click" (default) | "hover"
     open: reflects the panel state; set it to open or close from code (fires no event)
     value: key of the selected leaf (picker mode; omit for an action dropdown)
     items: leaf {key,label,description?,icon?,disabled?,danger?}
            · locked leaf {key,label,locked:true,plan?:"pro"|"essential",featureKey?,upgradeBody?}
              (shows the Paywall crown; click / Enter / Space opens the Paywall beside the row, never select)
            · submenu-parent {key,label,icon?,children:[…]}
            (opens an adjacent flyout — hover / ArrowRight / Enter; ArrowLeft / Escape closes it)
            · {type:"divider"} · {type:"group",label,children:[]} -->
<aha-dropdown id="menu" label="Actions" placement="bottomLeft"

<!-- A kebab menu: slot="trigger" swaps the default button for your own control (give an icon-only one an aria-label) -->
<aha-dropdown placement="bottomRight"
  items='[{"key":"rename","label":"Rename","icon":"system-pencil-simple"},{"key":"delete","label":"Delete","icon":"system-trash","danger":true}]'>
  <aha-button slot="trigger" variant="secondary" icon-only aria-label="More actions">
    <aha-icon slot="icon" name="system-dots-three-vertical" size="16" decorative></aha-icon>
  </aha-button>
</aha-dropdown>

<!-- value="editor" makes it a single-choice picker: the matching row is the brand-tinted selected one
     and the trigger shows its label (an explicit label wins). Choosing a row sets value and emits select. -->
<aha-dropdown id="role" value="editor"
  items='[
    {"key":"editor","label":"Editor","description":"Edit and present"},
    {"key":"presenter","label":"Presenter","description":"Present only, cannot edit"}
  ]'>
</aha-dropdown>

<script>
  document.getElementById('role').addEventListener('select', (e) => console.log('role', e.detail.key));
  // Opens on click, closes on outside-click / Escape (focus returns to the trigger); choosing an
  // item emits a composed `select`. Every user open and close emits `open` / `close`; set the `open`
  // attribute to drive it from code. The panel renders in the top layer, so a card cannot clip it.
  const menu = document.getElementById('menu');
  menu.addEventListener('select', (e) => console.log('action', e.detail.key));
  menu.addEventListener('open', () => console.log('menu opened'));
  menu.addEventListener('close', () => console.log('menu closed'));
  // A locked row's Paywall events bubble out of the dropdown; preventDefault() runs your own upgrade flow
  menu.addEventListener('paywall-upgrade', (e) => console.log('upgrade', e.detail.feature, e.detail.plan));
</script>
