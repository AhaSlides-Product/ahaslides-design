import '@ahaslides-product/design/aha-question-list'; // registers <aha-question-list>
import { useRef, useEffect } from 'react';

// React 18 needs a thin wrapper; React 19 can use <aha-question-list> directly.
// questions and notices are PROPERTIES; onChange gives the full edited model back plus the edit that made it.
// Writing questions back with the same ids updates in place, so the caret survives a controlled host.
function AhaQuestionList({ questions, notices, label, help, min, max, minOptions, maxOptions, allowImage, disabled, accordion, multiline, singleCorrect, testId, onChange, onToggle, onLeave, children }) {
  const ref = useRef();
  useEffect(() => {
    const el = ref.current;
    el.questions = questions || [];
    el.notices = notices || [];
    const h = (e) => onChange?.(e.detail.questions, e.detail.edit);   // composed CustomEvent
    const toggle = (e) => onToggle?.(e);   // cancelable: e.preventDefault() keeps the question as it was
    const leave = (e) => onLeave?.(e.detail);
    el.addEventListener('change', h);
    el.addEventListener('question-toggle', toggle);
    el.addEventListener('question-leave', leave);
    return () => { el.removeEventListener('change', h); el.removeEventListener('question-toggle', toggle); el.removeEventListener('question-leave', leave); };
  }, [questions, notices, onChange, onToggle, onLeave]);
  return (
    <aha-question-list
      ref={ref}
      {...(label ? { label } : {})}
      {...(help ? { help } : {})}
      {...(min != null ? { min: String(min) } : {})}
      {...(max != null ? { max: String(max) } : {})}
      {...(minOptions != null ? { minoptions: String(minOptions) } : {})}
      {...(maxOptions != null ? { maxoptions: String(maxOptions) } : {})}
      {...(allowImage ? { allowimage: '' } : {})}
      {...(disabled ? { disabled: '' } : {})}
      {...(accordion ? { accordion: '' } : {})}
      {...(multiline ? { multiline: '' } : {})}
      {...(singleCorrect ? { singlecorrect: '' } : {})}
      {...(testId ? { testid: testId } : {})}
    >{children}</aha-question-list>
  );
}
