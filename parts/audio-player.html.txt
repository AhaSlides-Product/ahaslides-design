<!-- Paste-and-run: save as .html and open in a browser. No build step.
     <aha-audio-player> is the SAME shared custom element React and Vue consume, here in its native form. -->
<link rel="stylesheet" href="https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/tokens.css">
<script type="module">
  import 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/aha-audio-player.js';   // registers <aha-audio-player>

  const player = document.getElementById('slide-audio');
  player.addEventListener('play', () => console.log('playing'));
  player.addEventListener('timeupdate', (e) => console.log(e.detail.currentTime, 'of', e.detail.duration));
  player.addEventListener('volumechange', (e) => console.log('volume', e.detail.volume, 'muted', e.detail.muted));
  player.addEventListener('ended', () => console.log('ended'));
</script>

<!-- one row from 400px wide -->
<aha-audio-player id="slide-audio" src="https://example.com/media/intro-music.mp3" track-title="intro-music.mp3"></aha-audio-player>
