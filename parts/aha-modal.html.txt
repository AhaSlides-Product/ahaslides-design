<script type="module">
  import 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/aha-modal.js';   // registers <aha-modal>
  import 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/aha-button.js';   // registers <aha-button>

  const modal = document.getElementById('delete-team');
  document.getElementById('open-delete').addEventListener('click', () => modal.show());

  // Async OK: keep it open, spin OK, close once the work succeeds.
  modal.addEventListener('ok', async (event) => {
    event.preventDefault();
    modal.setAttribute('confirm-loading', '');
    await new Promise((done) => setTimeout(done, 1000));
    modal.removeAttribute('confirm-loading');
    modal.open = false;
  });
  modal.addEventListener('cancel', (event) => console.log('cancelled from', event.detail.source));   // cancel-button, close-button, escape, mask
</script>
