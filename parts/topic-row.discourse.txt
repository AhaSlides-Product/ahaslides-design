<!-- 2. The theme's topic-list-item template: add the classes beside Discourse's own.

     Discourse markup                         add class / element
     tr.topic-list-item                       aha-topic-row
     tr.topic-list-item.visited               aha-topic-row--read (title drops to Regular)
     tr.topic-list-item.pinned                aha-topic-row--pinned, plus the pin icon below
     td (avatar cell)                         aha-topic-row__avatar, holding <aha-avatar size="40">
     td.main-link                             aha-topic-row__main
     span.link-top-line                       aha-topic-row__head
     a.title.raw-topic-link                   aha-topic-row__title (keep raw-topic-link)
     .topic-statuses .pinned (thumbtack)      <aha-icon class="aha-topic-row__pin" …> inside the title link
     a.badge-category__wrapper                <aha-tag href> for now (the category chip, G4, replaces it)
     div.topic-excerpt                        aha-topic-row__excerpt (plain text, 2 lines)
     likes / replies / activity              aha-topic-row__meta > aha-topic-row__stat (+ aha-topic-row__sr label)

     Row markup, Glimmer (the values are the theme's own topic fields): -->
<tr class="topic-list-item aha-topic-row {{if @topic.visited 'aha-topic-row--read'}} {{if @topic.pinned 'aha-topic-row--pinned'}}">
  <td class="aha-topic-row__avatar">
    <aha-avatar size="40" name={{@topic.creator.username}} src={{avatarUrl}} aria-hidden="true"></aha-avatar>
  </td>
  <td class="main-link topic-list-data aha-topic-row__main">
    <span class="link-top-line aha-topic-row__head">
      <a href={{@topic.lastUnreadUrl}} class="title raw-link raw-topic-link aha-topic-row__title" data-topic-id={{@topic.id}}>{{#if @topic.pinned}}<aha-icon class="aha-topic-row__pin" name="system-push-pin" size="16" label="Pinned"></aha-icon>{{/if}}{{@topic.title}}</a>
      <aha-tag href={{@topic.category.url}}>{{@topic.category.name}}</aha-tag>
    </span>
    <p class="topic-excerpt aha-topic-row__excerpt">{{plainExcerpt}}</p>
    <div class="aha-topic-row__meta">
      <span>{{@topic.creator.name}}</span>
      <span class="aha-topic-row__stat"><aha-icon name="system-heart-straight" size="16" decorative></aha-icon>{{@topic.like_count}}<span class="aha-topic-row__sr"> likes</span></span>
      <span class="aha-topic-row__stat"><aha-icon name="system-chat-circle" size="16" decorative></aha-icon>{{@topic.replyCount}}<span class="aha-topic-row__sr"> replies</span></span>
      <time datetime={{@topic.bumped_at}}>{{relativeAge}}</time>
    </div>
  </td>
</tr>
