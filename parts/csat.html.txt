<!-- Paste-and-run: save as .html and open in a browser. No build step.
     <aha-csat> is the SAME shared custom element React and Vue consume — here in its native
     form. Its thumbs are the shared <aha-icon>, and a thumbs-down opens a feedback popover built
     from the shared <aha-popover> + <aha-counted-textarea> + <aha-button> — all loaded by the
     element itself. Theming comes only from the --aha-* tokens in tokens.css. -->
<link rel="stylesheet" href="https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/tokens.css">
<script type="module">
  import 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/aha-csat.js';   // registers <aha-csat> (+ <aha-icon>, <aha-popover>, <aha-counted-textarea>, <aha-button>)
</script>

<!-- The borderless single row (the only layout): prompt + thumbs + the optional Feedback button on one line, no chrome.
     Thumbs-up rates instantly; thumbs-down opens the free-text feedback popover. -->
<aha-csat prompt="How did this import go?" source="docs_footer" feedback-button></aha-csat>

<script>
  const csat = document.querySelector('aha-csat');
  csat.addEventListener('rate', (e) => {
    console.log('CSAT', e.detail.rating, 'from', e.detail.source);   // { rating: 'up' | 'down' | null, source }
  });
  csat.addEventListener('feedback', (e) => {
    console.log('CSAT feedback', e.detail.feedback, 'from', e.detail.source);   // { rating: 'up' | 'down' | null, source, feedback }
  });
</script>
