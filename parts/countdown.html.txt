<!-- Paste-and-run: save as .html and open it. No build step, nothing to copy.
     In a Developer Platform slide type, applyDeck() gives every audience element the deck's ink, accent
     and font from the host's slide props (the host's --aha-* variables never reach the iframe). -->
<link rel="stylesheet" href="https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/tokens.css">
<script type="module">
  import 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/aha-countdown.js';   // registers <aha-countdown>
  import { applyDeck } from 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/audience-deck.js';
  applyDeck(document.documentElement, window.xprops?.slide ?? { textColour: '#1A1A1A', presentationColorPalette: ['#E70E68'] });
  const timer = document.querySelector('aha-countdown');
  timer.setAttribute('ends-at', String(Date.now() + 30000));   // the host's end instant, in a real slide
</script>

<aha-countdown total-ms="30000" urgent-below-ms="5000" label="Time remaining"></aha-countdown>
