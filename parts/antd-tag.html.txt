<!-- Paste-and-run: save as .html and open in a browser. No build step, no bundler.
     antd Tag is a COMPOSITE, so its HTML form is a CDN-React page: React and antd load from a CDN
     (esm.sh) and mount on open, using the SAME shared `tagTheme` the React and Vue wrappers use.
     New UI uses <aha-tag> instead. No JSX: React.createElement. -->
<div id="root"></div>
<script type="module">
  import React from 'https://esm.sh/react@18';
  import { createRoot } from 'https://esm.sh/react-dom@18/client';
  import { ConfigProvider, Tag } from 'https://esm.sh/antd@6?deps=react@18,react-dom@18';
  import { tagTheme } from 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/tag-theme.js';
  import 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/tag-theme.js';
  import 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/aha-button.js';   // registers <aha-button>

  const h = React.createElement;

  createRoot(document.getElementById('root')).render(
    h(ConfigProvider, { theme: tagTheme },
      h('div', { style: { display: 'flex', gap: 8 } }, h(Tag, null, '3 pages'), h(Tag, null, 'Quiz'))
    )
  );
</script>
