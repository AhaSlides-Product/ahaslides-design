     <aha-post> is one post of a forum post stream, the SAME element React and Vue consume. It reuses
     <aha-avatar>, <aha-tag> and <aha-button>. Theming comes only from the --aha-* tokens in tokens.css.
     The body is the element's children: pass sanitised HTML, never user input as raw markup. -->
<link rel="stylesheet" href="https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/tokens.css">
<script type="module">
  import 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/icons.js';
  import 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/aha-post.js';  // registers <aha-post>
  document.addEventListener('aha-post-action', (e) => {
    const post = e.target;
    if (e.detail.action !== 'like') return;
    const liked = !post.hasAttribute('liked');
    post.toggleAttribute('liked', liked);
    post.setAttribute('likes', String(Number(post.getAttribute('likes') || 0) + (liked ? 1 : -1)));
  });
</script>
