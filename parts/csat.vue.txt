// main.ts — register the element + mark aha-* as custom elements
import '@ahaslides-product/design/aha-csat';
app.config.compilerOptions.isCustomElement = (tag) => tag.startsWith('aha-');

// Component.vue — the custom element's `rate` event binds natively with @rate
<script setup>
function onRate(e) {
  track('CSAT_RATED', e.detail);   // { rating: 'up' | 'down', source }
}
</script>

<template>
  <!-- Default — the borderless single row: prompt + 16px thumbs on one line, no chrome -->
  <aha-csat prompt="How's your experience?" source="editor_help" @rate="onRate" />
  <!-- `card` → the boxed form: 40px bordered thumb buttons with a semibold prompt -->
  <aha-csat card prompt="Was this helpful?" source="editor_card" @rate="onRate" />
</template>
