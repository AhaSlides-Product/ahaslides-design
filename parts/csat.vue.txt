// main.ts — register the element + mark aha-* as custom elements
import '@ahaslides-product/design/aha-csat';
app.config.compilerOptions.isCustomElement = (tag) => tag.startsWith('aha-');

// Component.vue — the custom element's `rate` and `feedback` events bind natively with @rate / @feedback
<script setup>
function onRate(e) {
  track('CSAT_RATED', e.detail);   // { rating: 'up' | 'down' | null, source }
}
function onFeedback(e) {
  track('CSAT_FEEDBACK_SUBMITTED', e.detail);   // { rating: 'up' | 'down' | null, source, feedback }
}
</script>

<template>
  <!-- The borderless single row (the default layout). Thumbs-up rates instantly; thumbs-down opens the
       free-text feedback popover, which emits `feedback` on submit.
       tooltip-placement="bottom" opens the hover tooltips below (default top) — use it inside a titled card. -->
  <aha-csat prompt="How did this import go?" source="editor_help" feedback-button tooltip-placement="bottom" @rate="onRate" @feedback="onFeedback" />
  <!-- In a narrow side panel: layout="stacked" puts the prompt above the buttons; feedback-placement="start"
       opens the feedback popover start-aligned (extending right of the thumbs). -->
  <aha-csat prompt="How did this import go?" source="side_panel" feedback-button layout="stacked" feedback-placement="start" @rate="onRate" @feedback="onFeedback" />
  <!-- feedback-on="any" also opens the popover after a thumbs-up (default: thumbs-down only); confirmation="..." shows
       that text beside the thumbs once rated and again after Send (under them with layout="stacked"). -->
  <aha-csat prompt="How did this import go?" source="more_feedback" feedback-on="any" confirmation="Thanks, feedback received" @rate="onRate" @feedback="onFeedback" />
  <!-- feedback-hint="..." shows a helper line under the feedback text box, beside the counter. -->
  <aha-csat prompt="How did this import go?" source="hint" feedback-hint="What you write goes to the developer" @rate="onRate" @feedback="onFeedback" />
</template>
