import '@ahaslides-product/design/aha-answer-option';
import '@ahaslides-product/design/aha-bottom-sheet';
import { useEffect, useRef, useState } from 'react';

export function LongOption({ text, image }) {
  const option = useRef(null);
  const sheet = useRef(null);
  const [open, setOpen] = useState(false);
  useEffect(() => {
    const row = option.current, panel = sheet.current;
    const expand = () => setOpen(true), close = () => setOpen(false);
    row.addEventListener('expand', expand);
    panel.addEventListener('close', close);
    return () => { row.removeEventListener('expand', expand); panel.removeEventListener('close', close); };
  }, []);
  return (
    <>
      <aha-answer-option ref={option} expandable="" image={image}>{text}</aha-answer-option>
      <aha-bottom-sheet ref={sheet} open={open ? '' : undefined} image={image}>{text}</aha-bottom-sheet>
    </>
  );
}
