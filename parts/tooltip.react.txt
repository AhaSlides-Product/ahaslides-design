import '@ahaslides-product/design/aha-tooltip';   // registers <aha-tooltip>

// Attributes + slotted trigger. React < 19 does not bind custom events, so open-change goes through a ref.
import { useEffect, useRef } from 'react';
// placement: top | bottom | left | right (centred) OR edge-aligned top-start | top-end |
//   bottom-start | bottom-end | left-start | left-end | right-start | right-end
//   (a preference: with no room inside its panel or the viewport it opens on the side that has room)
// color: dark | brand · arrow: true|false · trigger: hover|focus|click|hover-tap (adds a touch tap) · help: built-in ? trigger
// helpLabel: the ? button's accessible name ("More info" when unset) · onOpenChange(open): every show and hide
function AhaTooltip({ text, placement, color, arrow, trigger, help, helpLabel, onOpenChange, children }) {
  const ref = useRef(null);
  useEffect(() => {
    const element = ref.current;
    if (!element || !onOpenChange) return undefined;
    const listener = (event) => onOpenChange(event.detail.open);
    element.addEventListener('open-change', listener);
    return () => element.removeEventListener('open-change', listener);
  }, [onOpenChange]);
  return (
    <aha-tooltip ref={ref} text={text} placement={placement} color={color} arrow={arrow} trigger={trigger}
      help={help ? '' : undefined} help-label={helpLabel}>
      {children}
    </aha-tooltip>
  );
}

// usage — wrap the focusable trigger; never hide essential info in a tooltip
// edge-aligned: same side as `bottom`, aligned to the trigger's start edge, arrow near it
<AhaTooltip text="Copy share link" placement="bottom-start" color="brand">
  <button aria-label="Copy share link">Share</button>
</AhaTooltip>

// help — the built-in `?` trigger (DS <aha-icon>, 16px --aha-icon-muted → --aha-icon-default on hover/focus); no slotted child needed,
// focusable so it shows on focus, not hover-only. The settings-label help-glyph pattern.
<label style={{ display: 'inline-flex', alignItems: 'center', gap: 6 }}>
  Shuffle options
  <AhaTooltip help text="Randomise the order options appear in for each participant" />
</label>

// hover-tap: a touch tap toggles it too · a named ? button · analytics on reveal
<AhaTooltip help trigger="hover-tap" helpLabel="About custom survey URLs"
  text="Share a memorable link instead of the generated code"
  onOpenChange={(open) => { if (open) track('tooltip_revealed'); }} />
