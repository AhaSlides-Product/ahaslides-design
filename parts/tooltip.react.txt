import '@ahaslides-product/design/aha-tooltip';   // registers <aha-tooltip>

// Attributes + slotted trigger only — no events — so React 18/19 can render it directly.
// placement: top | bottom | left | right · color: dark | brand · arrow: true|false · trigger: hover|focus|click
function AhaTooltip({ text, placement, color, arrow, trigger, children }) {
  return (
    <aha-tooltip text={text} placement={placement} color={color} arrow={arrow} trigger={trigger}>
      {children}
    </aha-tooltip>
  );
}

// usage — wrap the focusable trigger; never hide essential info in a tooltip
<AhaTooltip text="Copy share link" placement="bottom" color="brand">
  <button aria-label="Copy share link">Share</button>
</AhaTooltip>
