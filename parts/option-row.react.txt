import '@ahaslides-product/design/aha-option-row'; // registers <aha-option-row>
import { useRef, useEffect } from 'react';

// React 18 needs a thin wrapper; React 19 can use <aha-option-row> directly.
// Composes DS leaves; forwards input / correct-change / delete / reorder-move / reorder-grab / reorder-activate
// (+ the image control's intents). reorderable makes the drag handle a tab stop — the row never moves itself, the host does.
function AhaOptionRow({ value, maxLength, orderable, reorderable, correctable, correct, checkboxShape, singleLine, image, canDelete, onInput, onCorrect, onDelete, onReorderMove, onReorderGrab, onReorderActivate }) {
  const ref = useRef();
  useEffect(() => {
    const el = ref.current;
    el.value = value ?? '';
    if (correct) el.setAttribute('correct', ''); else el.removeAttribute('correct');
    const hi = (e) => onInput?.(e.detail.value);
    const hc = (e) => onCorrect?.(e.detail.correct);
    const hd = (e) => onDelete?.(e.detail.source);
    const hm = (e) => onReorderMove?.(e.detail.direction);
    const hg = (e) => onReorderGrab?.(e.detail);
    const ha = () => onReorderActivate?.();
    el.addEventListener('input', hi); el.addEventListener('correct-change', hc); el.addEventListener('delete', hd);
    el.addEventListener('reorder-move', hm); el.addEventListener('reorder-grab', hg); el.addEventListener('reorder-activate', ha);
    return () => {
      el.removeEventListener('input', hi); el.removeEventListener('correct-change', hc); el.removeEventListener('delete', hd);
      el.removeEventListener('reorder-move', hm); el.removeEventListener('reorder-grab', hg); el.removeEventListener('reorder-activate', ha);
    };
  });
  return (
    <aha-option-row
      ref={ref}
      {...(maxLength ? { maxlength: String(maxLength) } : {})}
      {...(orderable ? { orderable: '' } : {})}
      {...(reorderable ? { reorderable: '' } : {})}
      {...(correctable ? { correctable: '' } : {})}
      {...(checkboxShape ? { checkboxshape: checkboxShape } : {})}
      {...(singleLine ? { singleline: '' } : {})}
      {...(image ? { image: '' } : {})}
      {...(canDelete === false ? { candelete: 'false' } : {})}
    />
  );
}

// usage — one per option; disable delete at the minimum count. onDelete gets 'keyboard' (Backspace in an empty
// field) or 'pointer' (the delete button): after a 'keyboard' delete, call previousRow.focus(). onReorderMove
// gets 'up' | 'down' from the focused handle: move the row, then call rowEl.focusHandle() to restore focus.
{options.map((o, i) => (
  <AhaOptionRow key={o.id} value={o.text} maxLength={80} orderable reorderable correctable image
    checkboxShape="circle" correct={o.correct} canDelete={options.length > 2}
    onInput={(v) => update(i, v)} onCorrect={(c) => setCorrect(i, c)} onDelete={(source) => remove(i, source)}
    onReorderMove={(direction) => moveOption(i, direction)} />
))}
