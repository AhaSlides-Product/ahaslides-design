<!-- Paste-and-run: save as .html and open it. No build step, nothing to copy.
     In a Developer Platform slide type, applyDeck() gives every audience element the deck's ink, accent
     and font from the host's slide props (the host's --aha-* variables never reach the iframe). -->
<link rel="stylesheet" href="https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/tokens.css">
<script type="module">
  import 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/aha-audience-image-upload.js';   // registers <aha-audience-image-upload>
  import { applyDeck } from 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/audience-deck.js';
  applyDeck(document.documentElement, window.xprops?.slide ?? { textColour: '#1A1A1A', presentationColorPalette: ['#E70E68'] });
  const upload = document.querySelector('aha-audience-image-upload');
  upload.uploadImage = window.xprops?.uploadImage;   // the host's picker; absent here, so the control stays disabled
  upload.addEventListener('change', (event) => console.log('image', event.detail.value));
  upload.addEventListener('error', (event) => console.warn('upload failed', event.detail.error));
</script>

<aha-audience-image-upload add-label="Upload an image" hint="PNG, JPG, GIF, WebP"></aha-audience-image-upload>
