import '@ahaslides-product/design/aha-dropdown';  // registers <aha-dropdown>
import { useRef, useEffect } from 'react';

// React 18 needs a thin wrapper for the composed `select`, `change`, `open` and `close` events; React 19 can bind them directly.
// `open` + onOpenChange make it controlled; setting `open` fires no event, so the two cannot loop.
// onToggle(key, checked) receives a switch row's `change`; pass the new `checked` back in items to control it.
// onUpgrade receives a locked row's paywall-upgrade event (detail { feature, plan }); call e.preventDefault() to run your own flow.
function AhaDropdown({ label, value, items, placement, trigger, open, onSelect, onToggle, onOpenChange, onUpgrade }) {
  const ref = useRef();
  useEffect(() => {
    const el = ref.current;
    const h = (e) => onSelect?.(e.detail.key);
    const toggled = (e) => onToggle?.(e.detail.key, e.detail.checked);
    const opened = () => onOpenChange?.(true);
    const closed = () => onOpenChange?.(false);
    const upgrade = (e) => onUpgrade?.(e);
    el.addEventListener('select', h);
    el.addEventListener('change', toggled);
    el.addEventListener('open', opened);
    el.addEventListener('close', closed);
    el.addEventListener('paywall-upgrade', upgrade);
    return () => { el.removeEventListener('select', h); el.removeEventListener('change', toggled); el.removeEventListener('open', opened); el.removeEventListener('close', closed); el.removeEventListener('paywall-upgrade', upgrade); };
  });
  // Set the property, not the attribute: React 18 would write open="false" (read as closed) and React 19 assigns the property itself.
  useEffect(() => { if (open !== undefined) ref.current.open = !!open; }, [open]);
  return <aha-dropdown ref={ref} label={label} value={value} placement={placement} trigger={trigger} items={JSON.stringify(items)} />;
}
