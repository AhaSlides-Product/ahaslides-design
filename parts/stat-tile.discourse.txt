//      <link rel="stylesheet" href="https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/tokens.css">
//      <link rel="stylesheet" href="https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/stat-strip.css">
//
//    Class mapping (lib/stat-strip.css):
//      .about__stats                      → .aha-stat-strip         up to 5 a row, wraps, never scrolls sideways
//      .about__stats-item                 → .aha-stat-strip__item   plain tile; the figure and its word stay one string
//      .about__stats-item .d-icon-users, -shield-halved, -calendar-days → system-users, system-shield-check, system-calendar-dots
//      .about-stats-table                 → scrolls inside its own box on a narrow screen
//    Profile summary stats (.user-stat) are styled by lib/discourse-profile.css, not this file.
//
// 2. Optional, for a SemiBold figure on the About page: Discourse renders each About stat as one string
//    ("450 Members"), so CSS cannot set the figure apart. This initializer rebuilds the strip as
//    <aha-stat-tile> elements. javascripts/discourse/api-initializers/aha-about-stats.js, plus in the head:
//      <script type="module" src="https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/aha-stat-tile.js"></script>
import { apiInitializer } from "discourse/lib/api";

const FIGURE = /^\s*([\d.,]+[kKmM]?)\s+(.+?)\s*$/;

function rebuild(strip) {
  if (strip.dataset.ahaTiles) return;
  const tiles = [...strip.querySelectorAll(".about__stats-item")].map((item) => {
    const match = FIGURE.exec(item.textContent);
    if (!match) return null;
    const tile = document.createElement("aha-stat-tile");
    tile.setAttribute("value", match[1]);
    tile.setAttribute("label", match[2]);
    return tile;
  });
  if (!tiles.length || tiles.includes(null)) return;
  strip.dataset.ahaTiles = "1";
  strip.replaceChildren(...tiles);
}

export default apiInitializer("1.0", (api) => {
  api.onPageChange(() => document.querySelectorAll(".about__stats").forEach(rebuild));
});
