<!-- Paste-and-run: save as .html and open in a browser. No build step.
     <aha-import-progress> mirrors the presenter's Import notification: you own the task, the copy and the
     auto-dismiss. Update the same element in place through its states — never add a card per state. -->
<link rel="stylesheet" href="https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/tokens.css">
<script type="module">
  import 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/icons.js';                // <aha-icon> glyphs
  import 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/aha-import-progress.js';   // registers <aha-import-progress> (+ <aha-button>)

  const card = document.querySelector('aha-import-progress');
  let percent = 0;
  const timer = setInterval(() => {
    percent += 10;
    card.setAttribute('value', percent);
    card.setAttribute('caption', percent + '%');
    if (percent >= 100) {
      clearInterval(timer);
      card.setAttribute('state', 'success');
      card.setAttribute('caption', '40/40 slides imported');
      card.querySelector('[slot="action"]')?.remove();
      setTimeout(() => card.remove(), 2000);
    }
  }, 600);
  card.addEventListener('dismiss', () => card.remove());
  card.querySelector('[slot="action"]').addEventListener('click', () => {
    clearInterval(timer);
    card.setAttribute('state', 'canceled');
    card.setAttribute('heading', 'Import canceled');
    card.setAttribute('description', 'No slides were imported');
    card.setAttribute('caption', '');
    card.querySelector('[slot="action"]').remove();
  });
</script>

<aha-import-progress heading="Quarterly review.pptx" value="0" caption="0%" dismiss-label="Dismiss">
  <span slot="action" role="button" tabindex="0">Cancel</span>
</aha-import-progress>
