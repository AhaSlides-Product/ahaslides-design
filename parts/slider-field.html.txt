<!-- Paste-and-run: save as .html and open in a browser. No build step.
     <aha-slider> is the SAME shared custom element React and Vue consume, here in its native form. -->
<link rel="stylesheet" href="https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/tokens.css">
<script type="module">
  import 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/aha-slider.js';         // registers <aha-slider>
  import 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/aha-input-number.js';   // registers <aha-input-number>

  // valueText formats the value label and the spoken value
  const zoom = document.getElementById('zoom');
  zoom.valueText = (value) => `${Math.round(value * 100)}%`;
  zoom.addEventListener('input', (e) => console.log('zooming:', e.detail.value));    // antd onChange
  zoom.addEventListener('change', (e) => console.log('zoomed:', e.detail.value));    // antd onChangeComplete

  // a slider and a number field on the same value
  const overlay = document.getElementById('overlay');
  const overlayNumber = document.getElementById('overlay-number');
  overlay.addEventListener('input', (e) => { overlayNumber.value = e.detail.value; });
  overlayNumber.addEventListener('input', (e) => { if (e.detail.value != null) overlay.value = e.detail.value; });
</script>

<!-- min / max / step: a decimal step snaps exactly; the visible label names the handle -->
<span id="zoom-label">Zoom</span>
<aha-slider id="zoom" min="1" max="3" step="0.01" value="1" aria-labelledby="zoom-label"></aha-slider>

<!-- with an InputNumber field when the exact number matters: a 1fr 88px grid -->
<span id="overlay-label">Overlay</span>
<div style="display:grid;grid-template-columns:1fr 88px;gap:8px;align-items:center">
  <aha-slider id="overlay" min="0" max="100" value="30" aria-labelledby="overlay-label"></aha-slider>
  <aha-input-number id="overlay-number" min="0" max="100" value="30" aria-labelledby="overlay-label" style="width:100%"></aha-input-number>
</div>

<!-- states: disabled · no value label -->
<aha-slider disabled value="60" aria-label="Volume"></aha-slider>
<aha-slider tooltip="false" value="20" aria-label="Opacity"></aha-slider>
