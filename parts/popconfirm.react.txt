import { ConfigProvider, Popconfirm } from 'antd';   // antd v6
import { popconfirmTheme, destructiveConfirm } from '@ahaslides-product/design/popconfirm-theme';
import '@ahaslides-product/design/aha-button';        // registers <aha-button>
import '@ahaslides-product/design/icons';             // registers <aha-icon>

<ConfigProvider theme={popconfirmTheme}>
  <Popconfirm
    title="Delete this slide?"
    description="This cannot be undone."
    okText="Delete"
    cancelText="Cancel"
    placement="top"          // top | bottom | left | right — arrow tracks the trigger
    icon={<aha-icon name="system-warning-circle" size="16" style={{ marginInlineEnd: 8 }} />}
    {...destructiveConfirm}  // destructive: Cancel is the primary, Delete the dark danger button
    onConfirm={remove}
  >
    <aha-button variant="secondary" size="md">Delete slide</aha-button>
  </Popconfirm>
</ConfigProvider>
// One shared popconfirmTheme → the DS V3 look, action buttons at the DS small size (28px).
// The safe choice is the primary: spread destructiveConfirm on Delete / Remove; leave it off for a
// plain confirm (Publish, Apply), where the confirm is the Vivid Pink primary beside an outlined Cancel.
// Matrix: placement (top/bottom/left/right) × destructive (on/off).
// For an irreversible action use a Modal instead (overlays pattern).
