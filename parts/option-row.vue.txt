// main.ts — register the element + mark aha-* as custom elements
import '@ahaslides-product/design/aha-option-row';   // registers <aha-option-row>
app.config.compilerOptions.isCustomElement = (tag) => tag.startsWith('aha-');

// Component.vue
<script setup>
import { ref } from 'vue';
const options = ref([{ id: 1, text: 'Paris', correct: true }, { id: 2, text: 'London', correct: false }]);
// source is 'keyboard' (Backspace in an empty field) or 'pointer' (the delete button): after a 'keyboard' delete, call previousRow.focus()
function remove(i, source) { options.value.splice(i, 1); }
// reorderable makes the drag handle a tab stop; the row never moves itself. direction is 'up' | 'down' from
// ArrowUp/ArrowDown on the focused handle — move the row, then call the row element's focusHandle() to restore focus.
function move(i, direction) { /* swap options.value[i] with its neighbour, then e.target.focusHandle() */ }
// Enter/Space on the focused handle: the host decides what activating the grip should do.
function activate(i) {}
</script>

<template>
  <!-- one row per option; delete disables (not hides) at the minimum count -->
  <aha-option-row
    v-for="(o, i) in options"
    :key="o.id"
    orderable
    reorderable
    correctable
    image
    checkboxshape="circle"
    colordot
    :colorindex="(i % 6) + 1"
    :value.prop="o.text"
    maxlength="80"
    :correct="o.correct || undefined"
    :candelete="options.length > 2 ? 'true' : 'false'"
    @input="o.text = $event.detail.value"
    @correct-change="o.correct = $event.detail.correct"
    @delete="remove(i, $event.detail.source)"
    @reorder-move="move(i, $event.detail.direction)"
    @reorder-activate="activate(i)"
  />
</template>
