import '@ahaslides-product/design/aha-countdown';

export function QuestionTimer({ endsAt, totalMs, stopped }) {
  return <aha-countdown ends-at={endsAt} total-ms={totalMs} stopped={stopped ? '' : undefined} urgent-below-ms="5000" />;
}

export function PointsBar({ endsAt, totalMs, stopped }) {
  return <aha-countdown variant="points" min="500" max="1000" label-format="{n} pts" spoken-format="{n} points" ends-at={endsAt} total-ms={totalMs} stopped={stopped ? '' : undefined} />;
}
