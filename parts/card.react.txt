import '@ahaslides-product/design/aha-card';   // registers <aha-card>
import '@ahaslides-product/design/icons';      // registers <aha-icon>

// The custom element renders its own shadow DOM; React just passes attributes + slots.
// Boolean attrs (hoverable/interactive/stretch/loading) render as presence; string attrs (size/bordered) pass through.
// card-title is plain text; pass a richTitle node to fill slot="title" instead.
function DeckCard({ title, richTitle, size, bordered, hoverable, interactive, stretch, loading, onClick, cover, extra, actions, children }) {
  return (
    <aha-card
      card-title={title}
      size={size}
      bordered={bordered}
      {...(hoverable ? { hoverable: '' } : {})}
      {...(interactive ? { interactive: '' } : {})}
      {...(stretch ? { stretch: '' } : {})}
      {...(loading ? { loading: '' } : {})}
      onClick={onClick}
    >
      {cover && <img slot="cover" src={cover} alt="" />}
      {richTitle && <span slot="title">{richTitle}</span>}
      {extra && <a slot="extra" href="#">{extra}</a>}
      {children}
      {actions && <div slot="actions">{actions}</div>}
    </aha-card>
  );
}

// usage
<DeckCard title="Live poll" hoverable cover="cover.jpg" extra="Edit"
  actions={<><aha-icon name="system-share-network" size="16" /><aha-icon name="system-bookmark-simple" size="16" /></>}>
  Which release cadence works best for your team?
</DeckCard>
