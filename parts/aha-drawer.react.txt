import '@ahaslides-product/design/aha-drawer';
import '@ahaslides-product/design/aha-button';
import { useEffect, useRef, useState } from 'react';

// React (< 19) binds custom-element events through a ref; `open` is set as an attribute.
export function ParticipantReport() {
  const [open, setOpen] = useState(false);
  const drawer = useRef(null);
  useEffect(() => {
    const element = drawer.current;
    const onClosed = () => setOpen(false);
    element.addEventListener('closed', onClosed);
    return () => element.removeEventListener('closed', onClosed);
  }, []);
  return (
    <>
      <aha-button variant="secondary" onClick={() => setOpen(true)}>Participant report</aha-button>
      <aha-drawer ref={drawer} heading="Participant report" open={open ? '' : undefined}>
        <p>Minh Anh answered 4 of 5 questions correctly.</p>
        <div slot="footer" style={{ display: 'flex', justifyContent: 'flex-end' }}>
          <aha-button variant="primary" onClick={() => drawer.current.close()}>Done</aha-button>
        </div>
      </aha-drawer>
    </>
  );
}

// Bottom sheet: placement="bottom" is the same element. A click on the gap, Esc, the ✕ or a swipe down emits `close`.
export function DeckSheet({ deck, onClose }) {
  const sheet = useRef(null);
  useEffect(() => {
    const element = sheet.current;
    element.addEventListener('closed', onClose);
    return () => element.removeEventListener('closed', onClose);
  }, [onClose]);
  return (
    <aha-drawer ref={sheet} placement="bottom" heading={deck.title} open="">
      <span slot="subtitle">{deck.slideCount} slides · Updated {deck.updated}</span>
      <div slot="actions"><aha-button variant="secondary" size="sm">Edit</aha-button></div>
      <p>Slides and the Start list go here; the layout is yours.</p>
      <div slot="footer" style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
        <aha-button variant="tertiary" size="sm">Previous</aha-button>
        <span>3 of 24</span>
        <aha-button variant="tertiary" size="sm">Next</aha-button>
      </div>
    </aha-drawer>
  );
}
