<!-- Paste-and-run: save as .html and open in a browser. No build step, no bundler.
     DatePicker is a COMPOSITE (antd, no framework-free element), so its HTML form is a
     CDN-React page — React + antd load from a CDN (esm.sh) and mount on open. It
     consumes the SAME shared `datepickerTheme` the React/Vue wrappers use, so it renders
     the DS V3 look (36px field, radius 8, #E3E3E3 border, brand-selected cell).
     No JSX here (JSX needs a compiler) — we use React.createElement via the `h` alias. -->
<div id="root"></div>
<script type="module">
  import React from 'https://esm.sh/react@18';
  import { createRoot } from 'https://esm.sh/react-dom@18/client';
  import { ConfigProvider, DatePicker } from 'https://esm.sh/antd@6?deps=react@18,react-dom@18';
  import { datepickerTheme } from 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/datepicker-theme.js';

  const h = React.createElement;
  const { RangePicker } = DatePicker;

  createRoot(document.getElementById('root')).render(
    h(ConfigProvider, { theme: datepickerTheme },
      h('div', { style: { display: 'flex', flexDirection: 'column', gap: 16 } },
        h(DatePicker, { placeholder: 'Pick a date', style: { width: 240 } }),
        h(RangePicker, { style: { width: 280 } })
      )
    )
  );
</script>
