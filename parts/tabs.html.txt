<!-- Paste-and-run: save as .html and open in a browser. No build step.
     <aha-tabs> is the SAME shared custom element React and Vue consume.
     Each panel child carries its label in data-tab; it emits a composed `change` event. -->
<link rel="stylesheet" href="https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/tokens.css">
<script type="module">
  import 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/aha-tabs.js';   // registers <aha-tabs>
</script>

<aha-tabs id="tabs" value="0">
  <section data-tab="Overview">Overview panel content.</section>
  <section data-tab="Activity">Activity panel content.</section>
  <section data-tab="Settings">Settings panel content.</section>
</aha-tabs>

<script>
  // Plain DOM — no framework. Read e.detail.value (index) off the composed CustomEvent.
  document.getElementById('tabs').addEventListener('change', (e) => {
    console.log('active tab:', e.detail.value);
  });
</script>
