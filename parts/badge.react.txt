import '@ahaslides-product/design/aha-badge';   // registers <aha-badge>
import '@ahaslides-product/design/aha-paywall';
import '@ahaslides-product/design/icons';       // registers <aha-icon>

// Attributes only — no events — so React 18/19 can render the element directly.
// Boolean attrs (dot/showZero) are the attribute PRESENCE, so spread them conditionally.
function AhaBadge({ count, overflowCount, dot, status, text, color, showZero, ribbon, placement,
                    plan, cycle, session, kind, size, children }) {
  return (
    <aha-badge
      count={count}
      overflow-count={overflowCount}
      status={status}
      text={text}
      color={color}
      placement={placement}
      plan={plan}
      cycle={cycle}
      session={session}
      kind={kind}
      size={size}
      {...(dot ? { dot: '' } : {})}
      {...(showZero ? { 'show-zero': '' } : {})}
      {...(ribbon ? { ribbon: '' } : {})}
    >
      {children}
    </aha-badge>
  );
}

// New and Beta tags (size sm): <AhaBadge kind="new" /> · <AhaBadge kind="beta" />
// Answer results (Report app, size sm): <AhaBadge kind="correct" /> · <AhaBadge kind="incorrect" />

// count over a wrapped child
<AhaBadge count={unread} overflowCount={99}>
  <BellIcon />
</AhaBadge>

// a bare dot — status default · disabled · active (a dot on a bell is always active)
<AhaBadge dot status="active" />

// a corner ribbon banner wrapping a card (placement start/end)
<AhaBadge ribbon text="New">
  <TemplateCard />
</AhaBadge>

// a PLAN preset badge (default 28px, or size="sm" 20px; cycle appends the billing cycle)
<AhaBadge plan="pro" cycle="yearly" />
<AhaBadge plan="pro" size="sm" />

// a paid-feature marker: <aha-paywall> with no children is the Vivid Pink crown circle
<aha-paywall feature-label="Custom logo" required-plan="pro" />

// a SESSION badge (size sm by default)
<AhaBadge session={3} />
