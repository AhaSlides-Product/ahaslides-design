import '@ahaslides-product/design/aha-chart';   // registers <aha-chart>
import { useEffect, useRef } from 'react';

// Data is a JS property: set it through a ref so live updates re-use the drawn marks (they animate).
// type: bar | column | stacked | donut | radial | treemap | quadrant | bell | radar | wordcloud | mindmap
function ResultsChart({ type = 'bar', results, options, onCanvas, deckPalette, deckTextColour, locale, strings }) {
  const ref = useRef(null);
  useEffect(() => { ref.current.data = results; }, [results]);
  useEffect(() => { if (options) ref.current.options = options; }, [options]);   // merged over chartOptionDefaults[type]
  useEffect(() => { if (strings) ref.current.strings = strings; }, [strings]);   // the Presenter app's translations

  return (
    <aha-chart ref={ref} type={type} locale={locale}
      palette={onCanvas ? 'deck' : 'brand'}
      colors={onCanvas ? deckPalette.join(',') : undefined}
      ink={onCanvas ? deckTextColour : undefined}
      label="Poll results" />
  );
}
