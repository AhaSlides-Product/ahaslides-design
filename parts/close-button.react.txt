import '@ahaslides-product/design/aha-close-button';   // registers <aha-close-button> (+ <aha-button>, <aha-icon>)

// React 18/19 render the element directly; a press is a normal click.
// Alert, Info box and Background task already render it; Toast and Notification take it through
// dismissibleToast / dismissibleTitle, Modal and Drawer through dismissibleModalTitle /
// dismissibleDrawerTitle. Use it directly only for a new dismissible surface.
export function Notice({ children, onDismiss }) {
  return (
    <div style={{ display: 'flex', gap: 8, alignItems: 'flex-start', padding: '12px 16px' }}>
      <span>{children}</span>
      <aha-close-button
        label="Dismiss"
        onClick={onDismiss}
        style={{ '--aha-close-line-height': '21px', '--aha-close-container-padding': '16px' }}
      />
    </div>
  );
}
