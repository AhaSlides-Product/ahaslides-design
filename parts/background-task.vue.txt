// main.ts — register the elements + mark aha-* as custom elements
import '@ahaslides-product/design/aha-background-task';
import '@ahaslides-product/design/icons';
app.config.compilerOptions.isCustomElement = (tag) => tag.startsWith('aha-');

// ImportToast.vue
<script setup>
defineProps({ fileName: String, percent: Number, state: { type: String, default: 'progress' } });
const emit = defineEmits(['cancel', 'dismiss']);
</script>

<template>
  <aha-background-task :state="state" :heading="fileName" :value="percent" :caption="`${percent}%`"
    @dismiss="emit('dismiss')">
    <span v-if="state === 'progress'" slot="action" role="button" tabindex="0" @click="emit('cancel')">Cancel</span>
  </aha-background-task>
</template>
