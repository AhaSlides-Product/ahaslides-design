import { useEffect, useRef } from 'react';
import '@ahaslides-product/design/aha-csat';   // registers <aha-csat> (+ <aha-icon>)

// The element emits a composed `rate` event — attach a ref listener (React <19 pattern).
function AhaCsat({ prompt, source, card, onRate }) {
  const ref = useRef(null);
  useEffect(() => {
    const el = ref.current;
    const h = (e) => onRate?.(e.detail);   // { rating: 'up' | 'down', source }
    el.addEventListener('rate', h);
    return () => el.removeEventListener('rate', h);
  }, [onRate]);
  // `card` is a boolean attribute — pass "" to set it, omit it otherwise.
  return <aha-csat ref={ref} prompt={prompt} source={source} card={card ? '' : undefined} />;
}

// usage — the default borderless single row
<AhaCsat prompt="How's your experience?" source="editor_help" onRate={(d) => track('CSAT_RATED', d)} />
// boxed card form
<AhaCsat card prompt="Was this helpful?" source="editor_card" onRate={(d) => track('CSAT_RATED', d)} />
