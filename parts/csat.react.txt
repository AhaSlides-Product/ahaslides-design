import { useEffect, useRef } from 'react';
import '@ahaslides-product/design/aha-csat';   // registers <aha-csat> (+ <aha-icon>, <aha-popover>, <aha-counted-textarea>, <aha-button>)

// The element emits composed `rate` and `feedback` events — attach ref listeners (React <19 pattern).
function AhaCsat({ prompt, source, tooltipPlacement, layout, feedbackPlacement, feedbackOn, confirmation, feedbackHint, onRate, onFeedback }) {
  const ref = useRef(null);
  useEffect(() => {
    const el = ref.current;
    const rate = (e) => onRate?.(e.detail);         // { rating: 'up' | 'down' | null, source }
    const feedback = (e) => onFeedback?.(e.detail);  // { rating: 'up' | 'down' | null, source, feedback }
    el.addEventListener('rate', rate);
    el.addEventListener('feedback', feedback);
    return () => { el.removeEventListener('rate', rate); el.removeEventListener('feedback', feedback); };
  }, [onRate, onFeedback]);
  // tooltipPlacement: 'top' (default) | 'bottom' — open the tooltips below inside a titled card
  // layout: 'row' (default) | 'stacked' — prompt above the buttons; feedbackPlacement: 'end' (default) | 'start' — open the popover start-aligned
  // feedbackOn: 'down' (default) | 'any' — 'any' also opens the popover after a thumbs-up; confirmation: text shown beside the thumbs once rated and after Send; feedbackHint: helper line under the feedback text box
  return <aha-csat ref={ref} prompt={prompt} source={source} feedback-button tooltip-placement={tooltipPlacement} layout={layout} feedback-placement={feedbackPlacement} feedback-on={feedbackOn} confirmation={confirmation} feedback-hint={feedbackHint} />;
}

// usage — thumbs-up rates instantly; thumbs-down opens the free-text feedback popover (feedbackOn="any" opens it after a thumbs-up too)
<AhaCsat
  prompt="How did this import go?"
  source="editor_help"
  tooltipPlacement="bottom"
  layout="stacked"
  feedbackPlacement="start"
  feedbackOn="any"
  confirmation="Thanks, feedback received"
  feedbackHint="What you write goes to the developer"
  onRate={(d) => track('CSAT_RATED', d)}
  onFeedback={(d) => track('CSAT_FEEDBACK_SUBMITTED', d)}
/>
