<!-- Paste-and-run: save as .html and open in a browser. No build step.
     <aha-background-task> is the pattern for any long-running process; this example follows the presenter's Import notification: you own the task, the copy and the
     dismissal. Update the same element in place through its states — never add a card per state. -->
<link rel="stylesheet" href="https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/tokens.css">
<script type="module">
  import 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/icons.js';                // <aha-icon> glyphs
  import 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/aha-background-task.js';   // registers <aha-background-task> (+ <aha-button>, <aha-close-button>)

  const card = document.querySelector('aha-background-task');
  let percent = 0;
  const timer = setInterval(() => {
    percent += 10;
    card.setAttribute('value', percent);
    card.setAttribute('caption', percent + '%');
    if (percent >= 100) {
      clearInterval(timer);
      card.setAttribute('state', 'success');
      card.setAttribute('caption', '40/40 slides imported');
      card.querySelector('[slot="action"]')?.remove();
    }
  }, 600);
  card.addEventListener('dismiss', () => card.remove());
  card.querySelector('[slot="action"]').addEventListener('click', () => {
    clearInterval(timer);
    card.setAttribute('state', 'canceled');
    card.setAttribute('heading', 'Import cancelled');
    card.setAttribute('description', 'No slides were imported');
    card.setAttribute('caption', '');
    card.querySelector('[slot="action"]').remove();
  });
</script>

<aha-background-task heading="Quarterly review.pptx" value="0" caption="0%" dismiss-label="Dismiss">
  <aha-button slot="action" variant="tertiary" size="sm">Cancel</aha-button>
</aha-background-task>
