<!-- Paste-and-run: save as .html and open in a browser. No build step.
     <aha-loader> is the full-surface branded loading SCREEN for an environment transition
     (workspace → editor, editor → presenting): the animated AhaSlides splash, centred on white.
     One import registers it. Make its container full-screen for a real transition. -->
<script type="module">
  import 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/aha-loader.js';   // registers <aha-loader> (+ <aha-illustration> for the tiles variant)
</script>

<!-- Fill the viewport while the next environment boots. -->
<div style="position:fixed; inset:0">
  <aha-loader label="Opening the editor…"></aha-loader>
</div>

<!-- Or drop it into any sized box: the logo stays --aha-loader-size wide (default 112px), centred; min-height 240px. -->
<div style="width:100%; height:240px">
  <aha-loader label="Loading the settings…" style="--aha-loader-size:96px"></aha-loader>
</div>

<!-- Hold the screen until the app is ready AND the splash has played in full once. -->
<script type="module">
  const loader = document.querySelector('aha-loader');
  await Promise.all([loader.cycleDone, fetch('/api/ready')]);   // cycleDone: one full cycle (AhaLoader.CYCLE_MS = 3652)
  loader.remove();
</script>

<!-- Short or inline loads: the finished logo alone, no dot ring; the box is exactly --aha-loader-size. -->
<aha-loader logo-only label="Loading" style="width:auto; height:auto; --aha-loader-size:24px"></aha-loader>

<!-- The previous five-tile look. To switch every loader on the page at once instead, set
     <html data-aha-loader="tiles"> before the loaders connect; a variant attribute wins over it. -->
<div style="width:100%; height:360px">
  <aha-loader variant="tiles" label="Starting the presentation…"></aha-loader>
</div>
