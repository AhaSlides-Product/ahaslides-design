import { useEffect, useRef } from 'react';
import '@ahaslides-product/design/aha-progress-toast';   // registers <aha-progress-toast> (+ <aha-button>)
import '@ahaslides-product/design/icons';                // <aha-icon> glyphs

// Attributes + slots only — React 18/19 render the element directly; `dismiss` is a DOM event.
function ImportToast({ fileName, percent, state, onCancel, onDismiss }) {
  const ref = useRef(null);
  useEffect(() => {
    const node = ref.current;
    node.addEventListener('dismiss', onDismiss);
    return () => node.removeEventListener('dismiss', onDismiss);
  }, [onDismiss]);

  return (
    <aha-progress-toast ref={ref} state={state} heading={fileName} value={percent} caption={`${percent}%`}>
      {state === 'progress' && <span slot="action" role="button" tabIndex={0} onClick={onCancel}>Cancel</span>}
    </aha-progress-toast>
  );
}
