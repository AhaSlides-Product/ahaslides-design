import '@ahaslides-product/design/aha-select';   // registers <aha-select>
import '@ahaslides-product/design/icons';        // <aha-icon> — the trailing caret
import { useRef, useEffect } from 'react';

// React 18 needs a thin wrapper (options is a property, not a string attribute); React 19 can use
// <aha-select> directly. Same element, same shadow-DOM CSS → byte-identical with Vue.
function AhaSelect({ value, options, placeholder, size, status, disabled, onChange }) {
  const ref = useRef();
  useEffect(() => {
    const el = ref.current;
    el.options = options || [];              // set the array as a property
    if (value != null) el.value = value;
    el.disabled = !!disabled;
    const h = (e) => onChange?.(e.detail.value);   // composed CustomEvent
    el.addEventListener('change', h);
    return () => el.removeEventListener('change', h);
  });
  return (
    <aha-select ref={ref} placeholder={placeholder} size={size} status={status} />
  );
}

// usage
<AhaSelect
  placeholder="Pick a theme"
  value={theme}
  options={[
    { label: 'Light', value: 'light' },
    { label: 'Dark', value: 'dark' },
  ]}
  onChange={setTheme}
/>

// Free-entry tags (mode="tags"): the list is `values`, and change carries e.detail.values.
function AhaTagsSelect({ values, placeholder, onChange }) {
  const ref = useRef();
  useEffect(() => {
    const el = ref.current;
    el.values = values || [];
    const h = (e) => onChange?.(e.detail.values);
    el.addEventListener('change', h);
    return () => el.removeEventListener('change', h);
  });
  return <aha-select ref={ref} mode="tags" separators=", ;" placeholder={placeholder} />;
}

// Server search: show-search + filter="off" keeps every option you set; loading shows the spinner.
// <aha-select show-search filter="off" loading={busy ? '' : undefined} ...> and listen for 'search' (e.detail.query).

// Pick several (antd mode="multiple"): the same wrapper shape, reading e.detail.values.
// One row (antd maxTagCount="responsive"): <aha-select mode="multiple" max-tag-count="responsive" ...>; chips that do not fit collapse into a +N tag.
// antd → aha: allowClear → allow-clear, showSearch → show-search, optionFilterProp="title" → filter-prop="title",
// maxCount → max-count, mode={allowNew ? 'tags' : 'multiple'} → mode={allowNew ? 'tags' : 'multiple'} with the same options.
function AhaMultiSelect({ values, options, mode = 'multiple', allowClear, showSearch, filterProp, maxCount, ariaLabel, onChange }) {
  const ref = useRef();
  useEffect(() => {
    const el = ref.current;
    el.options = options || [];              // option groups: { label, options: [...] }; a trailing count: suffix
    el.values = values || [];
    const h = (e) => onChange?.(e.detail.values);
    el.addEventListener('change', h);
    return () => el.removeEventListener('change', h);
  });
  return (
    <aha-select ref={ref} mode={mode} aria-label={ariaLabel}
      allow-clear={allowClear ? '' : undefined} show-search={showSearch ? '' : undefined}
      filter-prop={filterProp} max-count={maxCount} />
  );
}
