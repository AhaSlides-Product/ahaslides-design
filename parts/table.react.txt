// Expandable rows — a detail panel (or tree rows) with the DS chevron, never a hand-rolled expandIcon.
import React from 'react';
import { createExpandableRows } from '@ahaslides-product/design/table-expandable';

const { useExpandableRows, DetailSection } = createExpandableRows({ React });   // once, at module scope

function ReportTable({ rows, columns }) {
  const { tableProps, decorateColumns } = useExpandableRows({
    dataSource: rows,
    rowKey: 'key',
    expandAllToggle: true,
    renderDetail: (record) => (
      <DetailSection title="Answers">
        <Table rowKey="key" pagination={false} columns={answerColumns} dataSource={record.answers} />
      </DetailSection>
    ),
    // omit renderDetail for tree rows (children share the parent's columns)
    // onExpand={(expanded, record) => fetch…} + loadingKeys={[…]} for a lazily loaded panel
  });
  return (
    <ConfigProvider theme={tableTheme}>
      <Table rowKey="key" dataSource={rows} columns={decorateColumns(columns)} {...tableProps} />
    </ConfigProvider>
  );
}
