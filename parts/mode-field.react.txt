import '@ahaslides-product/design/aha-mode-field'; // registers <aha-mode-field>
import { useRef, useEffect } from 'react';

// React 18 needs a thin wrapper; React 19 can use <aha-mode-field> directly.
// modes is a PROPERTY; bodies are children tagged data-mode — the hidden one keeps its state.
function AhaModeField({ label, modes, mode, disabled, onChange, children }) {
  const ref = useRef();
  useEffect(() => {
    const el = ref.current;
    el.modes = modes || [];
    if (mode != null) el.mode = mode;
    el.disabled = !!disabled;
    const h = (e) => onChange?.(e.detail.mode);   // composed CustomEvent
    el.addEventListener('change', h);
    return () => el.removeEventListener('change', h);
  }, [modes, mode, disabled, onChange]);
  return <aha-mode-field ref={ref} label={label}>{children}</aha-mode-field>;
}

// usage
<AhaModeField
  label="Results"
  mode={resultsMode}
  modes={[{ value: 'auto', label: 'Automatic' }, { value: 'manual', label: 'Manual' }]}
  onChange={setResultsMode}
>
  <div data-mode="auto">Results appear automatically.</div>
  <div data-mode="manual">You reveal results manually.</div>
</AhaModeField>
