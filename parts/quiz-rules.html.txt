<!-- Paste-and-run: save as .html and open it. No build step, nothing to copy.
     In a Developer Platform slide type, applyDeck() gives every audience element the deck's ink, accent
     and font from the host's slide props (the host's --aha-* variables never reach the iframe). -->
<link rel="stylesheet" href="https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/tokens.css">
<script type="module">
  import 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/aha-quiz-rules.js';   // registers <aha-quiz-rules>
  import { applyDeck } from 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/audience-deck.js';
  applyDeck(document.documentElement, window.xprops?.slide ?? { textColour: '#1A1A1A', presentationColorPalette: ['#E70E68'] });
  const rules = document.querySelector('aha-quiz-rules');
  rules.setAttribute('ends-at', String(Date.now() + 5000));   // when the rules end, from the page's own clock
  rules.rules = [
    { icon: 'system-timer', text: '20 seconds to answer' },
    { icon: 'system-star', text: 'Answer faster to get more points' },
  ];
  rules.addEventListener('skip', () => { rules.hidden = true; });   // the page ends the rules and shows the question
</script>

<main style="display:flex;flex-direction:column;min-height:100vh;margin:0 16px">
  <aha-quiz-rules heading="Question 2 of 7" skip-label="Skip rules" skip-testid="audience-quiz-skip-rules"></aha-quiz-rules>
</main>
