<!-- Paste-and-run: save as .html and open in a browser. No build step, no bundler.
     antd Segmented is a COMPOSITE, so its HTML form is a CDN-React page: React and antd load from a CDN
     (esm.sh) and mount on open, using the SAME shared `segmentedTheme` the React and Vue wrappers use.
     New UI uses <aha-segmented> instead. No JSX: React.createElement. -->
<div id="root"></div>
<script type="module">
  import React from 'https://esm.sh/react@18';
  import { createRoot } from 'https://esm.sh/react-dom@18/client';
  import { ConfigProvider, Segmented } from 'https://esm.sh/antd@6?deps=react@18,react-dom@18';
  import { segmentedTheme } from 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/segmented-theme.js';
  import 'https://cdn.jsdelivr.net/gh/ahaslides-product/ahaslides-design@__REF__/lib/aha-button.js';   // registers <aha-button>

  const h = React.createElement;

  createRoot(document.getElementById('root')).render(
    h(ConfigProvider, { theme: segmentedTheme },
      h(Segmented, { options: ['Desktop', 'Tablet', 'Mobile'], defaultValue: 'Desktop' })
    )
  );
</script>
