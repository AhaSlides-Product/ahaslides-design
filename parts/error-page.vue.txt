// main.ts — register the elements + mark aha-* as custom elements
import '@ahaslides-product/design/aha-error-page';
import '@ahaslides-product/design/icons';
app.config.compilerOptions.isCustomElement = (tag) => tag.startsWith('aha-');

// ErrorView.vue
<script setup>
const props = defineProps({ code: String });
const emit = defineEmits(['retry']);
function track(event) { analytics.track('error_page_view', { code: event.detail.code }); }
</script>

<template>
  <aha-error-page :code="props.code" @error-page-view="track" @action="emit('retry')" />
</template>
